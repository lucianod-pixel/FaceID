// Código para colar no Google Apps Script (dentro da sua planilha) — VERSÃO 2 // Colunas esperadas na primeira aba: A = NOME, B = EMAIL, C = REGISTRO // // Este código faz duas coisas: // 1) "cadastrar": quando uma pessoa é cadastrada no site, cria (ou atualiza) a linha dela na planilha. // 2) "registrar": quando uma pessoa é reconhecida, grava data/hora em REGISTRO e envia um e-mail // para o endereço da coluna EMAIL. var FUSO = 'America/Sao_Paulo'; function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(10000); // evita que dois envios simultâneos se atropelem var dados = JSON.parse(e.postData.contents); var acao = dados.acao || 'registrar'; var nome = String(dados.nome || '').trim(); if (!nome || nome.length > 80) { return resposta('nome invalido'); } var planilha = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; if (acao === 'cadastrar') { return cadastrar(planilha, nome, String(dados.email || '').trim()); } return registrar(planilha, nome); } catch (erro) { return resposta('erro: ' + erro); } finally { lock.releaseLock(); } } // ---------- Cadastro ---------- function cadastrar(planilha, nome, email) { if (!emailValido(email)) { return resposta('email invalido'); } var linha = encontrarLinha(planilha, nome); if (linha > 0) { planilha.getRange(linha, 2).setValue(email); // já existe: só atualiza o e-mail return resposta('email atualizado'); } planilha.appendRow([nome, email, '']); // nova linha: NOME, EMAIL, REGISTRO vazio return resposta('cadastrado'); } // ---------- Reconhecimento ---------- function registrar(planilha, nome) { var linha = encontrarLinha(planilha, nome); if (linha < 0) { return resposta('nome nao encontrado'); } var agora = Utilities.formatDate(new Date(), FUSO, 'dd/MM/yyyy HH:mm:ss'); planilha.getRange(linha, 3).setValue(agora); // coluna C = REGISTRO var email = String(planilha.getRange(linha, 2).getValue()).trim(); if (emailValido(email)) { try { MailApp.sendEmail({ to: email, subject: 'Reconhecimento facial: ' + nome + ' identificada(o) em ' + agora, body: 'Olá!\n\n' + 'Informamos que ' + nome + ' foi identificada(o) pelo sistema de reconhecimento facial ' + 'no dia e horário abaixo:\n\n' + agora + '\n\n' + 'Esta é uma mensagem automática. Não é necessário responder.' }); } catch (erroEmail) { return resposta('registrado, mas e-mail falhou: ' + erroEmail); } } return resposta('ok'); } // ---------- Funções de apoio ---------- // Procura o nome na coluna A e devolve o número da linha (ou -1 se não achar) function encontrarLinha(planilha, nome) { var ultima = planilha.getLastRow(); if (ultima < 2) return -1; var nomes = planilha.getRange(2, 1, ultima - 1, 1).getValues(); var procurado = nome.trim().toLowerCase(); for (var i = 0; i < nomes.length; i++) { if (String(nomes[i][0]).trim().toLowerCase() === procurado) { return i + 2; } } return -1; } function emailValido(email) { return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email); } function resposta(texto) { return ContentService.createTextOutput(texto); } // Execute esta função UMA VEZ, manualmente, para dar ao script a permissão de enviar e-mails. function autorizar() { Logger.log('E-mails restantes hoje: ' + MailApp.getRemainingDailyQuota()); }
